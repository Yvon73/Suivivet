Fichier nac/models.py
python
Copier

from django.db import models
from django.core.validators import MinValueValidator, MaxValueValidator

class CategoryNAC(models.Model):
    nom = models.CharField(max_length=100, unique=True, verbose_name="Nom de la catégorie")
    description = models.TextField(blank=True, verbose_name="Description")
    image = models.ImageField(upload_to='nac/categories/', blank=True, null=True, verbose_name="Image")

    def __str__(self):
        return self.nom

    class Meta:
        verbose_name = "Catégorie NAC"
        verbose_name_plural = "Catégories NAC"
        ordering = ['nom']

class SubCategoryNAC(models.Model):
    nom = models.CharField(max_length=100, unique=True, verbose_name="Nom de la sous-catégorie")
    description = models.TextField(blank=True, verbose_name="Description")
    category = models.ForeignKey(CategoryNAC, on_delete=models.CASCADE, related_name='subcategories', verbose_name="Catégorie")
    image = models.ImageField(upload_to='nac/subcategories/', blank=True, null=True, verbose_name="Image")

    def __str__(self):
        return f"{self.category.nom} - {self.nom}"

    class Meta:
        verbose_name = "Sous-Catégorie NAC"
        verbose_name_plural = "Sous-Catégories NAC"
        ordering = ['nom']

class EspeceNAC(models.Model):
    nom = models.CharField(max_length=100, unique=True, verbose_name="Nom de l'espèce")
    description = models.TextField(blank=True, verbose_name="Description")

    def __str__(self):
        return self.nom

    class Meta:
        verbose_name = "Espèce NAC"
        verbose_name_plural = "Espèces NAC"
        ordering = ['nom']

class NAC(models.Model):
    nom = models.CharField(max_length=100, unique=True, verbose_name="Nom commun")
    nom_scientifique = models.CharField(max_length=100, blank=True, null=True, verbose_name="Nom scientifique")
    categorie = models.ForeignKey(CategoryNAC, on_delete=models.SET_NULL, null=True, verbose_name="Catégorie")
    sous_categorie = models.ForeignKey(SubCategoryNAC, on_delete=models.SET_NULL, null=True, verbose_name="Sous-Catégorie")
    espece = models.ForeignKey(EspeceNAC, on_delete=models.SET_NULL, null=True, verbose_name="Espèce")

    taille_min = models.FloatField(blank=True, null=True, verbose_name="Taille minimale (cm)")
    taille_max = models.FloatField(blank=True, null=True, verbose_name="Taille maximale (cm)")
    poids_min = models.FloatField(blank=True, null=True, verbose_name="Poids minimal (kg)")
    poids_max = models.FloatField(blank=True, null=True, verbose_name="Poids maximal (kg)")
    esperance_vie = models.CharField(max_length=50, blank=True, null=True, verbose_name="Espérance de vie")

    origine = models.CharField(max_length=100, blank=True, null=True, verbose_name="Origine")
    niveau_soin = models.CharField(
        max_length=20,
        choices=[
            ('FACILE', 'Facile'),
            ('MODERE', 'Modéré'),
            ('DIFFICILE', 'Difficile'),
        ],
        default='MODERE',
        verbose_name="Niveau de soin"
    )
    legislation_france = models.CharField(max_length=100, blank=True, null=True, verbose_name="Législation en France")
    prix_moyen = models.CharField(max_length=50, blank=True, null=True, verbose_name="Prix moyen")

    description = models.TextField(blank=True, verbose_name="Description")
    image = models.ImageField(upload_to='nac/', blank=True, null=True, verbose_name="Image")

    # Alimentation
    type_alimentation = models.CharField(
        max_length=50,
        choices=[
            ('HERBIVORE', 'Herbivore strict'),
            ('CARNIVORE', 'Carnivore strict'),
            ('OMNIVORE', 'Omnivore'),
            ('INSECTIVORE', 'Insectivore'),
            ('GRANIVORE', 'Granivore'),
            ('FILTREUR', 'Filtreur'),
        ],
        blank=True,
        null=True,
        verbose_name="Type d'alimentation"
    )
    aliments_principaux = models.JSONField(default=list, blank=True, verbose_name="Aliments principaux")
    aliments_a_eviter = models.JSONField(default=list, blank=True, verbose_name="Aliments à éviter")
    frequence_alimentation = models.CharField(max_length=200, blank=True, null=True, verbose_name="Fréquence d'alimentation")

    # Habitat
    type_habitat = models.CharField(max_length=100, blank=True, null=True, verbose_name="Type d'habitat")
    taille_habitat_min = models.CharField(max_length=50, blank=True, null=True, verbose_name="Taille minimale de l'habitat")
    volume_habitat_min = models.FloatField(blank=True, null=True, verbose_name="Volume minimal de l'habitat (litres)")
    accessoires = models.JSONField(default=list, blank=True, verbose_name="Accessoires recommandés")
    temperature = models.JSONField(default=dict, blank=True, verbose_name="Température")
    humidite = models.CharField(max_length=50, blank=True, null=True, verbose_name="Humidité")
    eclairage = models.CharField(max_length=100, blank=True, null=True, verbose_name="Éclairage")
    social = models.BooleanField(default=False, verbose_name="Sociable (peut vivre en groupe)")

    # Comportement
    temperament = models.JSONField(default=list, blank=True, verbose_name="Tempérament")
    bruit = models.CharField(max_length=50, blank=True, null=True, verbose_name="Niveau de bruit")
    activite = models.CharField(max_length=50, blank=True, null=True, verbose_name="Période d'activité")

    # Santé
    problemes_courants = models.JSONField(default=list, blank=True, verbose_name="Problèmes de santé courants")
    vaccins = models.JSONField(default=list, blank=True, verbose_name="Vaccins")
    vermifuges = models.CharField(max_length=100, blank=True, null=True, verbose_name="Fréquence des vermifuges")
    soins_veto = models.CharField(max_length=200, blank=True, null=True, verbose_name="Soins vétérinaires recommandés")

    # Reproduction
    maturite_sexuelle = models.CharField(max_length=50, blank=True, null=True, verbose_name="Maturité sexuelle")
    duree_gestation = models.CharField(max_length=50, blank=True, null=True, verbose_name="Durée de gestation/incubation")
    nombre_petits = models.CharField(max_length=50, blank=True, null=True, verbose_name="Nombre de petits/œufs")
    sevrage = models.CharField(max_length=50, blank=True, null=True, verbose_name="Durée du sevrage")
    conseils_reproduction = models.TextField(blank=True, verbose_name="Conseils pour la reproduction")

    # Conseils
    conseils = models.JSONField(default=list, blank=True, verbose_name="Conseils pour les propriétaires")

    # Législation
    legislation = models.JSONField(default=dict, blank=True, verbose_name="Législation par pays")

    def __str__(self):
        return f"{self.nom} ({self.categorie})"

    class Meta:
        verbose_name = "NAC"
        verbose_name_plural = "NAC"
        ordering = ['categorie', 'nom']




2. Créer le script d’import
Crée un fichier nac/management/commands/importer_nac.py :
python
Copier

from django.core.management.base import BaseCommand
from nac.models import CategoryNAC, SubCategoryNAC, EspeceNAC, NAC
import json
from django.conf import settings
import os

class Command(BaseCommand):
    help = 'Importe les NAC depuis un fichier JSON'

    def handle(self, *args, **options):
        json_file_path = os.path.join(settings.BASE_DIR, 'static', 'data', 'nac.json')

        try:
            with open(json_file_path, 'r', encoding='utf-8') as f:
                data = json.load(f)
        except FileNotFoundError:
            self.stdout.write(self.style.ERROR(f'Fichier {json_file_path} introuvable'))
            return
        except json.JSONDecodeError:
            self.stdout.write(self.style.ERROR('Fichier JSON invalide'))
            return

        # Importer les espèces
        for espece_data in data.get('especes', []):
            espece, created = EspeceNAC.objects.get_or_create(
                nom=espece_data['nom'],
                defaults={'description': espece_data.get('description', '')}
            )
            if created:
                self.stdout.write(self.style.SUCCESS(f'Espèce "{espece.nom}" créée'))
            else:
                self.stdout.write(self.style.WARNING(f'Espèce "{espece.nom}" existe déjà'))

        # Importer les catégories et sous-catégories
        for category_data in data.get('categories', []):
            category, created = CategoryNAC.objects.get_or_create(
                nom=category_data['nom'],
                defaults={'description': category_data.get('description', '')}
            )
            if created:
                self.stdout.write(self.style.SUCCESS(f'Catégorie "{category.nom}" créée'))
            else:
                self.stdout.write(self.style.WARNING(f'Catégorie "{category.nom}" existe déjà'))

            for subcategory_data in category_data.get('sous_categories', []):
                subcategory, created = SubCategoryNAC.objects.get_or_create(
                    nom=subcategory_data['nom'],
                    category=category,
                    defaults={'description': subcategory_data.get('description', '')}
                )
                if created:
                    self.stdout.write(self.style.SUCCESS(f'Sous-catégorie "{subcategory.nom}" créée'))
                else:
                    self.stdout.write(self.style.WARNING(f'Sous-catégorie "{subcategory.nom}" existe déjà'))

        # Importer les NAC
        for nac_data in data.get('nac', []):
            # Récupérer la catégorie et la sous-catégorie
            category = CategoryNAC.objects.filter(nom=nac_data.get('categorie')).first()
            subcategory = SubCategoryNAC.objects.filter(nom=nac_data.get('sous_categorie')).first() if nac_data.get('sous_categorie') else None
            espece = EspeceNAC.objects.filter(nom=nac_data.get('espece')).first()

            # Créer ou mettre à jour le NAC
            nac, created = NAC.objects.get_or_create(
                nom=nac_data['nom'],
                defaults={
                    'nom_scientifique': nac_data.get('nom_scientifique'),
                    'categorie': category,
                    'sous_categorie': subcategory,
                    'espece': espece,
                    'taille_min': nac_data.get('taille_min'),
                    'taille_max': nac_data.get('taille_max'),
                    'poids_min': nac_data.get('poids_min'),
                    'poids_max': nac_data.get('poids_max'),
                    'esperance_vie': nac_data.get('esperance_vie'),
                    'origine': nac_data.get('origine'),
                    'niveau_soin': nac_data.get('niveau_soin', 'MODERE'),
                    'legislation_france': nac_data.get('legislation_france'),
                    'prix_moyen': nac_data.get('prix_moyen'),
                    'description': nac_data.get('description', ''),
                    'type_alimentation': nac_data.get('alimentation', {}).get('type'),
                    'aliments_principaux': nac_data.get('alimentation', {}).get('aliments_principaux', []),
                    'aliments_a_eviter': nac_data.get('alimentation', {}).get('aliments_a_eviter', []),
                    'frequence_alimentation': nac_data.get('alimentation', {}).get('frequence'),
                    'type_habitat': nac_data.get('habitat', {}).get('type'),
                    'taille_habitat_min': nac_data.get('habitat', {}).get('taille_min'),
                    'volume_habitat_min': nac_data.get('habitat', {}).get('volume_min'),
                    'accessoires': nac_data.get('habitat', {}).get('accessoires', []),
                    'temperature': nac_data.get('habitat', {}).get('temperature', {}),
                    'humidite': nac_data.get('habitat', {}).get('humidite'),
                    'eclairage': nac_data.get('habitat', {}).get('eclairage'),
                    'social': nac_data.get('comportement', {}).get('social', False),
                    'temperament': nac_data.get('comportement', {}).get('tempérament', []),
                    'bruit': nac_data.get('comportement', {}).get('bruit'),
                    'activite': nac_data.get('comportement', {}).get('activite'),
                    'problemes_courants': nac_data.get('sante', {}).get('problemes_courants', []),
                    'vaccins': nac_data.get('sante', {}).get('vaccins', []),
                    'vermifuges': nac_data.get('sante', {}).get('vermifuges'),
                    'soins_veto': nac_data.get('sante', {}).get('soins_veto'),
                    'maturite_sexuelle': nac_data.get('reproduction', {}).get('maturite_sexuelle'),
                    'duree_gestation': nac_data.get('reproduction', {}).get('duree_gestation'),
                    'nombre_petits': nac_data.get('reproduction', {}).get('nombre_petits'),
                    'sevrage': nac_data.get('reproduction', {}).get('sevrage'),
                    'conseils_reproduction': nac_data.get('reproduction', {}).get('conseils', ''),
                    'conseils': nac_data.get('conseils', []),
                    'legislation': nac_data.get('legislation', {}),
                }
            )
            if created:
                self.stdout.write(self.style.SUCCESS(f'NAC "{nac.nom}" créé'))
            else:
                self.stdout.write(self.style.WARNING(f'NAC "{nac.nom}" existe déjà'))

        self.stdout.write(self.style.SUCCESS('Import des NAC terminé !'))

from django.views.generic import ListView
from django.db.models import Q
from .models import NAC, CategoryNAC

class NACListView(ListView):
    model = NAC
    template_name = 'nac/liste.html'
    context_object_name = 'nacs'
    paginate_by = 20

    def get_queryset(self):
        queryset = super().get_queryset()
        category_slug = self.kwargs.get('category_slug')
        subcategory_slug = self.kwargs.get('subcategory_slug')
        espece_slug = self.kwargs.get('espece_slug')

        if category_slug:
            category = CategoryNAC.objects.get(slug=category_slug)
            queryset = queryset.filter(categorie=category)
        if subcategory_slug:
            subcategory = SubCategoryNAC.objects.get(slug=subcategory_slug)
            queryset = queryset.filter(sous_categorie=subcategory)
        if espece_slug:
            espece = EspeceNAC.objects.get(slug=espece_slug)
            queryset = queryset.filter(espece=espece)

        # Filtres par recherche
        search_query = self.request.GET.get('q')
        if search_query:
            queryset = queryset.filter(
                Q(nom__icontains=search_query) |
                Q(nom_scientifique__icontains=search_query) |
                Q(origine__icontains=search_query)
            )

        return queryset.order_by('nom')

    def get_context_data(self, **kwargs):
        context = super().get_context_data(**kwargs)
        context['categories'] = CategoryNAC.objects.all().prefetch_related('subcategories')
        context['especes'] = EspeceNAC.objects.all()
        return context

from django.db import models
from django.utils.text import slugify
from django.urls import reverse

class CategoryNAC(models.Model):
    nom = models.CharField(max_length=100, unique=True, verbose_name="Nom de la catégorie")
    slug = models.SlugField(max_length=100, unique=True, blank=True, verbose_name="Slug")
    description = models.TextField(blank=True, verbose_name="Description")
    image = models.ImageField(upload_to='nac/categories/', blank=True, null=True, verbose_name="Image")

    def save(self, *args, **kwargs):
        if not self.slug:
            self.slug = slugify(self.nom)
        super().save(*args, **kwargs)

    def __str__(self):
        return self.nom

    class Meta:
        verbose_name = "Catégorie NAC"
        verbose_name_plural = "Catégories NAC"
        ordering = ['nom']

class SubCategoryNAC(models.Model):
    nom = models.CharField(max_length=100, unique=True, verbose_name="Nom de la sous-catégorie")
    slug = models.SlugField(max_length=100, unique=True, blank=True, verbose_name="Slug")
    description = models.TextField(blank=True, verbose_name="Description")
    category = models.ForeignKey(CategoryNAC, on_delete=models.CASCADE, related_name='subcategories', verbose_name="Catégorie")
    image = models.ImageField(upload_to='nac/subcategories/', blank=True, null=True, verbose_name="Image")

    def save(self, *args, **kwargs):
        if not self.slug:
            self.slug = slugify(self.nom)
        super().save(*args, **kwargs)

    def __str__(self):
        return f"{self.category.nom} - {self.nom}"

    class Meta:
        verbose_name = "Sous-Catégorie NAC"
        verbose_name_plural = "Sous-Catégories NAC"
        ordering = ['nom']

class EspeceNAC(models.Model):
    nom = models.CharField(max_length=100, unique=True, verbose_name="Nom de l'espèce")
    slug = models.SlugField(max_length=100, unique=True, blank=True, verbose_name="Slug")
    description = models.TextField(blank=True, verbose_name="Description")

    def save(self, *args, **kwargs):
        if not self.slug:
            self.slug = slugify(self.nom)
        super().save(*args, **kwargs)

    def __str__(self):
        return self.nom

    class Meta:
        verbose_name = "Espèce NAC"
        verbose_name_plural = "Espèces NAC"
        ordering = ['nom']

class NAC(models.Model):
    nom = models.CharField(max_length=100, unique=True, verbose_name="Nom commun")
    slug = models.SlugField(max_length=100, unique=True, blank=True, verbose_name="Slug")
    nom_scientifique = models.CharField(max_length=100, blank=True, null=True, verbose_name="Nom scientifique")
    categorie = models.ForeignKey(CategoryNAC, on_delete=models.SET_NULL, null=True, verbose_name="Catégorie")
    sous_categorie = models.ForeignKey(SubCategoryNAC, on_delete=models.SET_NULL, null=True, verbose_name="Sous-Catégorie")
    espece = models.ForeignKey(EspeceNAC, on_delete=models.SET_NULL, null=True, verbose_name="Espèce")

    # ... (autres champs)

    def save(self, *args, **kwargs):
        if not self.slug:
            self.slug = slugify(self.nom)
        super().save(*args, **kwargs)

    def __str__(self):
        return f"{self.nom} ({self.categorie})"

    def get_absolute_url(self):
        return reverse('nac:nac_detail', kwargs={'slug': self.slug})

    def get_niveau_soin_badge_color(self):
        if self.niveau_soin == 'FACILE':
            return 'success'
        elif self.niveau_soin == 'MODERE':

    class Meta:
        verbose_name = "NAC"

        ordering = ['categorie', 'nom']

Vue pour ajouter/supprimer des favoris
python

from django.http import JsonResponse
from django.contrib.auth.decorators import login_required
from .models import FavorisNAC

@login_required
def toggle_favoris(request, nac_id):
    nac = NAC.objects.get(pk=nac_id)
    favoris, created = FavorisNAC.objects.get_or_create(utilisateur=request.user, nac=nac)

    if not created:
        favoris.delete()
        is_favoris = False
    else:
        is_favoris = True

    return JsonResponse({'is_favoris': is_favoris})

<script>
document.addEventListener('DOMContentLoaded', function() {
    const favorisButtons = document.querySelectorAll('.toggle-favoris');

    favorisButtons.forEach(button => {
        button.addEventListener('click', function(e) {
            e.preventDefault();
            const nacId = this.getAttribute('data-nac-id');
            const icon = this.querySelector('i');

            fetch(`/nac/${nacId}/favoris/`, {
                method: 'POST',
                headers: {
                    'X-CSRFToken': '{{ csrf_token }}',
                    'Content-Type': 'application/json'
                },
                body: JSON.stringify({})
            })
            .then(response => response.json())
            .then(data => {
                if (data.is_favoris) {
                    icon.classList.remove('bi-heart');
                    icon.classList.add('bi-heart-fill');
                } else {
                    icon.classList.remove('bi-heart-fill');
                    icon.classList.add('bi-heart');
                }
            })
            .catch(error => console.error('Erreur:', error));
        });
    });
});
</script>

    utilisateur = models.ForeignKey(User, on_delete=models.CASCADE, related_name='avis_nac')
    nac = models.ForeignKey(NAC, on_delete=models.CASCADE, related_name='avis')
    note = models.PositiveSmallIntegerField(
        choices=[(1, '1 - Très difficile'), (2, '2 - Difficile'), (3, '3 - Modéré'), (4, '4 - Facile'), (5, '5 - Très facile')],
        verbose_name="Note"
    )
    commentaire = models.TextField(blank=True, verbose_name="Commentaire")
    date = models.DateTimeField(auto_now_add=True)

    def __str__(self):
        return f"{self.utilisateur.username} - {self.nac.nom} ({self.note}/5)"

    class Meta:
        verbose_name = "Avis NAC"
        verbose_name_plural = "Avis NAC"
        unique_together = ('utilisateur', 'nac')

from django.contrib.auth.decorators import login_required
from django.shortcuts import render, redirect, get_object_or_404
from .models import NAC, AvisNAC
from .forms import AvisNACForm

@login_required
def ajouter_avis(request, nac_id):
    nac = get_object_or_404(NAC, pk=nac_id)
    avis_existant = AvisNAC.objects.filter(utilisateur=request.user, nac=nac).first()

    if request.method == 'POST':
        form = AvisNACForm(request.POST)
        if form.is_valid():
            note = form.cleaned_data['note']
            commentaire = form.cleaned_data['commentaire']

            if avis_existant:
                avis_existant.note = note
                avis_existant.commentaire = commentaire
                avis_existant.save()
            else:
                AvisNAC.objects.create(
                    utilisateur=request.user,
                    nac=nac,
                    note=note,
                    commentaire=commentaire
                )

            return redirect('nac:nac_detail', slug=nac.slug)
    else:
        form = AvisNACForm(initial={
            'note': avis_existant.note if avis_existant else None,
            'commentaire': avis_existant.commentaire if avis_existant else None
        })

    return render(request, 'nac/ajouter_avis.html', {
        'nac': nac,
        'form': form,
        'avis_existant': avis_existant
    })
